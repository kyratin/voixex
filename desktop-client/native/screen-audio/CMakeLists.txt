cmake_minimum_required(VERSION 3.20)
project(voixex_screen_audio LANGUAGES CXX)
add_executable(voixex-screen-audio main.cpp)
target_compile_features(voixex-screen-audio PRIVATE cxx_std_17)
target_compile_definitions(voixex-screen-audio PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(voixex-screen-audio PRIVATE ole32 mmdevapi uuid)
if(MSVC)
  set_property(TARGET voixex-screen-audio PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(voixex-screen-audio PRIVATE /W4 /EHsc)
endif()
